function loadMyBookings() { const bookings = JSON.parse(localStorage.getItem('myBookings')) || []; const container = document.getElementById('bookings-list'); if (!container) return; if (bookings.length === 0) { container.innerHTML = "

No tickets booked yet.

"; return; } container.innerHTML = bookings.map(item => `

${item.eventName}

Booking ID: ${item.bookingId}

Event Date: ${item.date}

Booked On: ${item.bookingDate}

Price Paid: ₹${item.price}

`).join(''); } document.addEventListener('DOMContentLoaded', loadMyBookings);

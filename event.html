function loadEvents() { const events = JSON.parse(localStorage.getItem('events')) || []; const container = document.getElementById('event-container'); if(!container) return; container.innerHTML = events.map(event => `
${event.category}

${event.title}

Date: ${event.date}

Available Tickets: ${event.ticketsAvailable}

Price: ₹${event.price}

`).join(''); } // Page load par execute karein document.addEventListener('DOMContentLoaded', loadEvents);
